cmake_minimum_required(VERSION 3.25)
project(HuntReplayMedia VERSION 0.1.2 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Build with Visual Studio 2022 x64 and Windows SDK 10.0.22621+ on Windows.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(FFMPEG_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/vendor/ffmpeg" CACHE PATH "Pinned FFmpeg shared SDK")
find_path(FFMPEG_INCLUDE_DIR libavcodec/avcodec.h PATHS "${FFMPEG_ROOT}/include" NO_DEFAULT_PATH REQUIRED)
add_library(HuntReplay.Media SHARED src/api.cpp src/capture.cpp src/video.cpp src/audio.cpp src/devices.cpp src/mux.cpp)
target_include_directories(HuntReplay.Media PRIVATE include "${FFMPEG_INCLUDE_DIR}")
target_compile_definitions(HuntReplay.Media PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0A00 WINVER=0x0A00)
target_compile_options(HuntReplay.Media PRIVATE /W4 /EHsc /permissive- /utf-8 /Zc:__cplusplus)
target_link_libraries(HuntReplay.Media PRIVATE d3d11 dxgi dxguid windowsapp runtimeobject ole32 uuid propsys avrt)
foreach(component avcodec avformat avutil swresample swscale)
  find_library(FFMPEG_${component} NAMES ${component} PATHS "${FFMPEG_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
  target_link_libraries(HuntReplay.Media PRIVATE "${FFMPEG_${component}}")
endforeach()
set_target_properties(HuntReplay.Media PROPERTIES OUTPUT_NAME "HuntReplay.Media")
install(TARGETS HuntReplay.Media RUNTIME DESTINATION .)
